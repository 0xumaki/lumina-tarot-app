   657→/** Format seconds as a human-readable duration (e.g., "10:00", "1:30:00") */
   658→function formatDuration(sec: number): string {
   659→  if (sec < 60) return `${sec}s`;
   660→  const h = Math.floor(sec / 3600);
   661→  const m = Math.floor((sec % 3600) / 60);
   662→  const s = sec % 60;
   663→  if (h > 0) return `${h}h ${m}m`;
   664→  if (m >= 10) return `${m}m`;
   665→  return `${m}:${String(s).padStart(2, "0")}`;
   666→}
