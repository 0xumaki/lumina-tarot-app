   915→function ClearDataModal({ open, onOpenChange }: { open: boolean; onOpenChange: (o: boolean) => void }) {
   916→  const api = useApi();
   917→  const { toast } = useToast();
   918→  const [loading, setLoading] = React.useState(false);
   919→
   920→  async function clearData() {
   921→    setLoading(true);
   922→    try {
   923→      // Clear localStorage + reload to reset device id
   924→      localStorage.clear();
   925→      toast({ title: "Data cleared", description: "Starting fresh." });
   926→      setTimeout(() => window.location.reload(), 800);
   927→    } finally {
   928→      setLoading(false);
   929→    }
   930→  }
   931→
   932→  return (
   933→    <motion.div
   934→      initial={{ opacity: 0 }}
   935→      animate={{ opacity: open ? 1 : 0 }}
   936→      className={`fixed inset-0 z-[70] flex items-end sm:items-center justify-center ${open ? "" : "pointer-events-none"}`}
   937→      onClick={() => onOpenChange(false)}
   938→    >
   939→      {open && <div className="absolute inset-0 bg-black/80 backdrop-blur-md" />}
   940→      <motion.div
   941→        initial={{ y: 50, opacity: 0, scale: 0.97 }}
   942→        animate={{ y: open ? 0 : 50, opacity: open ? 1 : 0, scale: open ? 1 : 0.97 }}
   943→        transition={{ type: "spring", stiffness: 300, damping: 30 }}
   944→        className="relative w-full max-w-md m-3 lum-glass-float rounded-[28px] overflow-hidden"
   945→        onClick={(e) => e.stopPropagation()}
   946→      >
   947→        <div className="p-6 text-center">
   948→          <div className="w-14 h-14 rounded-full bg-destructive/15 border border-destructive/30 flex items-center justify-center mx-auto mb-4">
   949→            <AlertCircle className="w-7 h-7 text-destructive" />
   950→          </div>
   951→          <h2 className="text-[18px] font-medium text-ink">Clear all data?</h2>
   952→          <p className="text-[13px] text-ink-muted mt-2 leading-[18px] max-w-[280px] mx-auto">
   953→            This will erase all your readings, goals, moods, confirmations, and stats from this device. This cannot be undone.
   954→          </p>
   955→          <div className="mt-5 space-y-2">
   956→            <button
   957→              onClick={clearData}
   958→              disabled={loading}
   959→              className="w-full rounded-full py-3 text-[14px] font-medium bg-destructive text-white active:scale-[0.98] transition-all disabled:opacity-50"
   960→            >
   961→              {loading ? "Clearing…" : "Yes, erase everything"}
