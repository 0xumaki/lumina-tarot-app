   125→   125→/** Animate a number from its previous value to `target` over `duration` ms. */
   126→   126→function useCountUp(target: number, duration = 900): number {
   127→   127→  const [val, setVal] = React.useState(0);
   128→   128→  const prev = React.useRef(0);
   129→   129→
   130→   130→  React.useEffect(() => {
   131→   131→    if (target <= 0) {
   132→   132→      setVal(0);
   133→   133→      prev.current = 0;
   134→   134→      return;
   135→   135→    }
   136→   136→    let raf = 0;
   137→   137→    const start = performance.now();
   138→   138→    const from = prev.current;
   139→   139→    const delta = target - from;
   140→   140→
   141→   141→    const tick = (now: number) => {
   142→   142→      const t = Math.min(1, (now - start) / duration);
   143→   143→      // easeOutCubic
   144→   144→      const eased = 1 - Math.pow(1 - t, 3);
   145→   145→      const next = Math.round(from + delta * eased);
   146→   146→      setVal(next);
   147→   147→      prev.current = next;
   148→   148→      if (t < 1) raf = requestAnimationFrame(tick);
   149→   149→    };
   150→   150→
   151→   151→    raf = requestAnimationFrame(tick);
   152→   152→    return () => cancelAnimationFrame(raf);
   153→   153→  }, [target, duration]);
   154→   154→
   155→   155→  return val;
   156→   156→}

   198→   198→function ActivityChart({ data }: { data: StatsData["activity"] }) {
   199→   199→  // Normalise each metric to its own max so all three stay visible even
   200→   200→  // though readings/confirmations are small counts and frequency is minutes.
   201→   201→  const maxReadings = Math.max(1, ...data.map((d) => d.readings));
   202→   202→  const maxConf = Math.max(1, ...data.map((d) => d.confirmations));
   203→   203→  const maxFreqMin = Math.max(
   204→   204→    1,
   205→   205→    ...data.map((d) => Math.round(d.frequencySec / 60))
   206→   206→  );
   207→   207→
   208→   208→  const W = 700;
   209→   209→  const H = 200;
   210→   210→  const padTop = 6;
   211→   211→  const chartH = H - padTop;
   212→   212→  const colW = W / 7;
   213→   213→  const barW = 16;
   214→   214→  const gap = 5;
   215→   215→  const groupW = barW * 3 + gap * 2;
   216→   216→
   217→   217→  return (
   218→   218→    <div>
   219→   219→      <svg
   220→   220→        viewBox={`0 0 ${W} ${H}`}
   221→   221→        preserveAspectRatio="none"
   222→   222→        width="100%"
   223→   223→        height={120}
   224→   224→        className="block"
   225→   225→        role="img"
   226→   226→        aria-label="7-day activity chart"
   227→   227→      >
   228→   228→        {/* baseline */}
   229→   229→        <line
   230→   230→          x1={0}
   231→   231→          x2={W}
   232→   232→          y1={H}
   233→   233→          y2={H}
   234→   234→          stroke="rgba(255,255,255,0.06)"
   235→   235→          strokeWidth={1}
   236→   236→          vectorEffect="non-scaling-stroke"
   237→   237→        />
   238→   238→        {data.map((d, i) => {
   239→   239→          const cx = colW * i + colW / 2;
   240→   240→          const rH = (d.readings / maxReadings) * chartH;
   241→   241→          const cH = (d.confirmations / maxConf) * chartH;
   242→   242→          const fMin = Math.round(d.frequencySec / 60);
   243→   243→          const fH = (fMin / maxFreqMin) * chartH;
   244→   244→          const startX = cx - groupW / 2;
   245→   245→          const baseY = H;
   246→   246→          return (
   247→   247→            <g key={d.date}>
   248→   248→              {/* readings — gold */}
   249→   249→              <rect
   250→   250→                x={startX}
   251→   251→                y={baseY - rH}
   252→   252→                width={barW}
   253→   253→                height={rH}
   254→   254→                rx={2}
   255→   255→                fill={GOLD}
   256→   256→                opacity={d.readings > 0 ? 1 : 0.14}
   257→   257→              />
   258→   258→              {/* confirmations — leaf */}
   259→   259→              <rect
   260→   260→                x={startX + barW + gap}
   261→   261→                y={baseY - cH}
   262→   262→                width={barW}
   263→   263→                height={cH}
   264→   264→                rx={2}
   265→   265→                fill={LEAF}
   266→   266→                opacity={d.confirmations > 0 ? 1 : 0.14}
   267→   267→              />
   268→   268→              {/* frequency minutes — sage */}
   269→   269→              <rect
   270→   270→                x={startX + (barW + gap) * 2}
   271→   271→                y={baseY - fH}
   272→   272→                width={barW}
   273→   273→                height={fH}
   274→   274→                rx={2}
   275→   275→                fill={SAGE}
   276→   276→                opacity={fMin > 0 ? 0.7 : 0.12}
   277→   277→              />
   278→   278→            </g>
   279→   279→          );
   280→   280→        })}
   281→   281→      </svg>
   282→   282→
   283→   283→      {/* Day labels (HTML, to avoid SVG distortion) */}
   284→   284→      <div className="grid grid-cols-7 mt-1.5">
   285→   285→        {data.map((d) => {
   286→   286→          const day = new Date(d.date + "T00:00:00").getDay();
   287→   287→          return (
   288→   288→            <div
   289→   289→              key={d.date}
   290→   290→              className="text-center text-[10px] text-ink-muted font-medium"
   291→   291→            >
   292→   292→              {DAY_LABELS[day]}
   293→   293→            </div>
   294→   294→          );
   295→   295→        })}
   296→   296→      </div>
   297→   297→
   298→   298→      {/* Legend */}
   299→   299→      <div className="flex items-center justify-center gap-4 mt-3">
   300→   300→        <Legend color={GOLD} label="Readings" />
   301→   301→        <Legend color={LEAF} label="Confirmations" />
   302→   302→        <Legend color={SAGE} label="Frequency min" />
   303→   303→      </div>
   304→   304→    </div>
   305→   305→  );
   306→   306→}

   641→   641→function RefineInsight({ type }: { type: "energy_signature" | "weekly_reflection" }) {
   642→   642→  const api = useApi();

   649→   649→      await api("/api/feedback", {
   650→   650→        method: "POST",
   651→   651→        body: JSON.stringify({ type, rating: r }),

   654→   654→  }

   656→   656→  if (rating) {
   657→   657→    return (

   659→   659→        {rating === "up" ? "Thanks ✦" : "Noted — we'll refine"}
   660→   660→      </span>
   661→   661→    );
   662→   662→  }
   663→   663→
   664→   664→  return (
   665→   665→    <div className="flex items-center gap-1">
   666→   666→      <span className="text-[10px] text-ink-muted mr-1">Helpful?</span>
   667→   667→      <button
   668→   668→        onClick={() => submit("up")}

   670→   670→        aria-label="Thumbs up"

   672→   672→        <ThumbsUp className="w-3 h-3" />
   673→   673→      </button>
   674→   674→      <button
   675→   675→        onClick={() => submit("down")}

   677→   677→        aria-label="Thumbs down"
