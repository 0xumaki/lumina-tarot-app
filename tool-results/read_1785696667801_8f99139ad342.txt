   125→/** Animate a number from its previous value to `target` over `duration` ms. */
   126→function useCountUp(target: number, duration = 900): number {
   127→  const [val, setVal] = React.useState(0);
   128→  const prev = React.useRef(0);
   129→
   130→  React.useEffect(() => {
   131→    if (target <= 0) {
   132→      setVal(0);
   133→      prev.current = 0;
   134→      return;
   135→    }
   136→    let raf = 0;
   137→    const start = performance.now();
   138→    const from = prev.current;
   139→    const delta = target - from;
   140→
   141→    const tick = (now: number) => {
   142→      const t = Math.min(1, (now - start) / duration);
   143→      // easeOutCubic
   144→      const eased = 1 - Math.pow(1 - t, 3);
   145→      const next = Math.round(from + delta * eased);
   146→      setVal(next);
   147→      prev.current = next;
   148→      if (t < 1) raf = requestAnimationFrame(tick);
   149→    };
   150→
   151→    raf = requestAnimationFrame(tick);
   152→    return () => cancelAnimationFrame(raf);
   153→  }, [target, duration]);
   154→
   155→  return val;
   156→}

   196→const DAY_LABELS = ["Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"];

   198→function ActivityChart({ data }: { data: StatsData["activity"] }) {
   199→  // Normalise each metric to its own max so all three stay visible even
   200→  // though readings/confirmations are small counts and frequency is minutes.
   201→  const maxReadings = Math.max(1, ...data.map((d) => d.readings));
   202→  const maxConf = Math.max(1, ...data.map((d) => d.confirmations));
   203→  const maxFreqMin = Math.max(
   204→    1,
   205→    ...data.map((d) => Math.round(d.frequencySec / 60))
   206→  );
   207→
   208→  const W = 700;
   209→  const H = 200;
   210→  const padTop = 6;
   211→  const chartH = H - padTop;
   212→  const colW = W / 7;
   213→  const barW = 16;
   214→  const gap = 5;
   215→  const groupW = barW * 3 + gap * 2;
   216→
   217→  return (
   218→    <div>
   219→      <svg
   220→        viewBox={`0 0 ${W} ${H}`}
   221→        preserveAspectRatio="none"
   222→        width="100%"
   223→        height={120}
   224→        className="block"
   225→        role="img"
   226→        aria-label="7-day activity chart"
   227→      >
   228→        {/* baseline */}
   229→        <line
   230→          x1={0}
   231→          x2={W}
   232→          y1={H}
   233→          y2={H}
   234→          stroke="rgba(255,255,255,0.06)"
   235→          strokeWidth={1}
   236→          vectorEffect="non-scaling-stroke"
   237→        />
   238→        {data.map((d, i) => {
   239→          const cx = colW * i + colW / 2;
   240→          const rH = (d.readings / maxReadings) * chartH;
   241→          const cH = (d.confirmations / maxConf) * chartH;
   242→          const fMin = Math.round(d.frequencySec / 60);
   243→          const fH = (fMin / maxFreqMin) * chartH;
   244→          const startX = cx - groupW / 2;
   245→          const baseY = H;
   246→          return (
   247→            <g key={d.date}>
   248→              {/* readings — gold */}
   249→              <rect
   250→                x={startX}
   251→                y={baseY - rH}
   252→                width={barW}
   253→                height={rH}
   254→                rx={2}
   255→                fill={GOLD}
   256→                opacity={d.readings > 0 ? 1 : 0.14}
   257→              />
   258→              {/* confirmations — leaf */}
   259→              <rect
   260→                x={startX + barW + gap}
   261→                y={baseY - cH}
   262→                width={barW}
   263→                height={cH}
   264→                rx={2}
   265→                fill={LEAF}
   266→                opacity={d.confirmations > 0 ? 1 : 0.14}
   267→              />
   268→              {/* frequency minutes — sage */}
   269→              <rect
   270→                x={startX + (barW + gap) * 2}
   271→                y={baseY - fH}
   272→                width={barW}
   273→                height={fH}
   274→                rx={2}
   275→                fill={SAGE}
   276→                opacity={fMin > 0 ? 0.7 : 0.12}
   277→              />
   278→            </g>
   279→          );
   280→        })}
   281→      </svg>
   282→
   283→      {/* Day labels (HTML, to avoid SVG distortion) */}
   284→      <div className="grid grid-cols-7 mt-1.5">
   285→        {data.map((d) => {
   286→          const day = new Date(d.date + "T00:00:00").getDay();
   287→          return (
   288→            <div
   289→              key={d.date}
   290→              className="text-center text-[10px] text-ink-muted font-medium"
   291→            >
   292→              {DAY_LABELS[day]}
   293→            </div>
   294→          );
   295→        })}
   296→      </div>
   297→
   298→      {/* Legend */}
   299→      <div className="flex items-center justify-center gap-4 mt-3">
   300→        <Legend color={GOLD} label="Readings" />
   301→        <Legend color={LEAF} label="Confirmations" />
   302→        <Legend color={SAGE} label="Frequency min" />
   303→      </div>
   304→    </div>
   305→  );
   306→}

   329→function moodLabel(avg: number | null): string {
   330→  if (avg === null) return "—";
   331→  const idx = Math.min(4, Math.max(0, Math.round(avg) - 1));
   332→  return MOOD_LABELS[idx];

   665→    <div className="flex items-center gap-1">
   666→      <span className="text-[10px] text-ink-muted mr-1">Helpful?</span>

   674→      <button
   675→        onClick={() => submit("down")}

   677→        aria-label="Thumbs down"

   679→        <ThumbsDown className="w-3 h-3" />

  1271→function MoodHistoryChart({ days }: { days: { date: string; mood: number | null; note: string | null }[] }) {
  1272→  const W = 320;
  1273→  const H = 120;
  1274→  const padTop = 8;
  1275→  const padBottom = 16;
  1276→  const chartH = H - padTop - padBottom;
  1277→  const colW = W / days.length;
  1278→
  1279→  // Build the line path for connected mood points
  1280→  const points = days
  1281→    .map((d, i) => (d.mood !== null ? { x: colW * i + colW / 2, y: padTop + chartH - ((d.mood - 1) / 4) * chartH, mood: d.mood } : null))
  1282→    .filter((p): p is { x: number; y: number; mood: number } => p !== null);
  1283→
  1284→  const pathD = points.length > 1
  1285→    ? points.map((p, i) => `${i === 0 ? "M" : "L"} ${p.x.toFixed(1)} ${p.y.toFixed(1)}`).join(" ")
  1286→    : "";
  1287→
  1288→  return (
  1289→    <div>
  1290→      <svg viewBox={`0 0 ${W} ${H}`} width="100%" height={H} className="block">
  1291→        {/* Grid lines for mood levels 1-5 */}
  1292→        {[1, 2, 3, 4, 5].map((lvl) => {
  1293→          const y = padTop + chartH - ((lvl - 1) / 4) * chartH;
  1294→          return (
  1295→            <g key={lvl}>
  1296→              <line x1={0} x2={W} y1={y} y2={y} stroke="rgba(255,255,255,0.04)" strokeWidth={0.5} />
  1297→              <text x={2} y={y - 2} fill="rgba(122,134,128,0.5)" fontSize={7}>{lvl}</text>
  1298→            </g>
  1299→          );
  1300→        })}
  1301→        {/* Connecting line */}
  1302→        {pathD && (
  1303→          <path d={pathD} fill="none" stroke="#C5A87C" strokeWidth={1.5} strokeLinecap="round" strokeLinejoin="round" opacity={0.6} />
  1304→        )}
  1305→        {/* Points */}
  1306→        {points.map((p, i) => (
  1307→          <circle
  1308→            key={i}
  1309→            cx={p.x}
  1310→            cy={p.y}
  1311→            r={2.5}
  1312→            fill={MOOD_COLORS[Math.min(4, Math.max(0, p.mood - 1))]}
  1313→            stroke="#000"
  1314→            strokeWidth={0.5}
  1315→          />
  1316→        ))}
  1317→      </svg>
  1318→      {/* Day labels (every 5th day) */}
  1319→      <div className="flex justify-between mt-1 px-1">
  1320→        {days.map((d, i) => {
  1321→          if (i % 5 !== 0 && i !== days.length - 1) return null;
  1322→          const date = new Date(d.date + "T00:00:00");
  1323→          return (
  1324→            <span key={i} className="text-[8px] text-ink-muted">
  1325→              {date.toLocaleDateString(undefined, { month: "numeric", day: "numeric" })}
  1326→            </span>
  1327→          );
  1328→        })}
  1329→      </div>
  1330→    </div>
  1331→  );
  1332→}
